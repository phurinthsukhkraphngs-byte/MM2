local function saveCustomEmotesToDisk()
    pcall(function()
        if writefile and type(customEmotes) == "table" then
            local data = HttpService:JSONEncode(customEmotes)
            writefile(CUSTOM_EMOTE_FILE, data)
        end
    end)
end

local function loadCustomEmotesFromDisk()
    customEmotes = {}
    pcall(function()
        if isfile and readfile and isfile(CUSTOM_EMOTE_FILE) then
            local raw = readfile(CUSTOM_EMOTE_FILE)
            local decoded = HttpService:JSONDecode(raw)
            if type(decoded) == "table" then
                customEmotes = decoded
            end
        end
    end)
end

-- SAFE CLIPBOARD FUNCTION (CROSS-EXECUTOR)
--==================================================
local function safeSetClipboard(str)
    pcall(function()
        if setclipboard then
            setclipboard(str)
        elseif toclipboard then
            toclipboard(str)
        elseif syn and syn.write_clipboard then
            syn.write_clipboard(str)
        elseif Clipboard and Clipboard.set then
            Clipboard.set(str)
        end
    end)
end

        for i = 1, 8 do

            task.wait(0.018)

local function getRole(player)
    local role = "Innocent"
    if PlayerRoles[player.Name] then
        if PlayerRoles[player.Name].Dead then return "Dead" end
        role = PlayerRoles[player.Name].Role
    end

    local hasKnife, hasGun = false, false
    local function scanTools(parent)
        if not parent then return end
        for _, obj in ipairs(parent:GetChildren()) do
            if obj:IsA("Tool") then
                local n = obj.Name:lower()
                if n:find("knife") or n:find("stab") or obj:FindFirstChild("KnifeClient") then hasKnife = true

    if hasKnife then return "Murderer"
    elseif hasGun then return (role == "Sheriff" and "Sheriff" or "Hero") end

function getPredictedTarget(character, originPos)
    if not character then return nil, nil, false end

    -- 1. Prioritize UpperTorso (R15) or Torso (R6) for the largest and most reliable bullet hitbox!
    local targetPart = character:FindFirstChild("UpperTorso")
        or character:FindFirstChild("Torso")
        or character:FindFirstChild("HumanoidRootPart")
        or character:FindFirstChild("Head")
    if not targetPart then return nil, nil, false end

    local root = character:FindFirstChild("HumanoidRootPart") or targetPart
    local hum = character:FindFirstChildOfClass("Humanoid")

    -- 2. Base position: center of the chest/torso
    local basePos = targetPart.Position

    -- 3. True ground velocity calculation
    local vel = root.AssemblyLinearVelocity or root.Velocity or Vector3.zero
    if hum and hum.MoveDirection.Magnitude > 0 then
        local ws = (hum.WalkSpeed > 0) and hum.WalkSpeed or 16
        if vel.Magnitude < 2 or vel.Magnitude > 40 then
            vel = hum.MoveDirection * ws
        else
            local horiz = hum.MoveDirection * ws
            vel = Vector3.new(horiz.X, vel.Y, horiz.Z)
        end
    elseif vel.Magnitude < 1 then
        vel = Vector3.zero
    end

    -- 4. Calculate accurate one-way ping (RTT / 2) - avoids overshooting!
    local ping = 0.035
    pcall(function()
        local pingStat = LocalPlayer:GetNetworkPing()
        if pingStat and pingStat > 0 then
            ping = math.clamp(pingStat / 2, 0.015, 0.12)
        end
    end)

    -- 5. Distance and bullet lead prediction
    local dist = (basePos - originPos).Magnitude
    local bulletSpeed = 440 -- MM2 revolver projectile speed
    local travelTime = (dist / bulletSpeed) + ping

    -- Dampen vertical jump bounce to prevent shooting above head
    local horizontalVel = Vector3.new(vel.X, 0, vel.Z)
    local clampedY = math.clamp(vel.Y, -10, 15) * 0.4

    local predictedPos
    if horizontalVel.Magnitude < 1.0 then
        predictedPos = basePos
    else
        predictedPos = basePos + (horizontalVel * travelTime) + Vector3.new(0, clampedY * travelTime, 0)
    end

    -- 6. Raycast wall check: test line of sight
    local isWallBlocked = false
    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.FilterDescendantsInstances = { LocalPlayer.Character, character, ESPFolder }
    rayParams.IgnoreWater = true

    local rayResult = workspace:Raycast(originPos, (predictedPos - originPos), rayParams)
    if rayResult then
        isWallBlocked = true
        -- If chest is blocked, check if Head is peeking above cover
        local head = character:FindFirstChild("Head")
        if head then
            local headRay = workspace:Raycast(originPos, (head.Position - originPos), rayParams)
            if not headRay then
                predictedPos = head.Position
                isWallBlocked = false
            end
        end
    end

    local targetCF = CFrame.new(predictedPos, predictedPos + (horizontalVel.Magnitude > 0 and horizontalVel or root.CFrame.LookVector))
    return predictedPos, targetCF, isWallBlocked
end

function executeShoot(teleport)
    local murderer = findRole("Murderer")
    if not murderer or not murderer.Character then
        WindUI:Notify({ Title = "Combat", Content = (currentLang == "TH" and "ไม่พบฆาตกรที่ยังมีชีวิตอยู่!" or "No Murderer found alive!"), Icon = "alert-triangle" })
        return
    end
    
    local mChar = murderer.Character
    local mRoot = mChar:FindFirstChild("HumanoidRootPart") or mChar:FindFirstChild("Torso") or mChar:FindFirstChild("Head")
    if not mRoot then return end

    local char = LocalPlayer.Character
    local myRoot = char and char:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end

    local gun = char:FindFirstChild("Gun") or LocalPlayer.Backpack:FindFirstChild("Gun")
    if not gun then
        WindUI:Notify({ Title = "Combat", Content = (currentLang == "TH" and "กรุณาถือปืนก่อนใช้งาน!" or "Equip Gun first!"), Icon = "alert-circle" })
        return
    end

    local hum = char:FindFirstChildOfClass("Humanoid")
    if gun.Parent == LocalPlayer.Backpack and hum then
        hum:EquipTool(gun)
        local t = tick() + 0.4
        repeat task.wait() until char:FindFirstChild("Gun") or tick() > t
    end

    local shootRemote = getShootRemote()
    if not shootRemote then return end

    local oldCF = myRoot.CFrame

    if teleport then
        -- 1. TP SHOOT (Always Behind Murderer Dynamic Auto-Track Engine)
        local stickOffset = CFrame.new(0, 1.2, 3.2) -- 3.2 studs directly behind murderer
        local startTime = tick()
        local duration = 0.14 -- Active stick duration before returning

        -- Temporary Noclip so our character doesn't bump into murderer or walls
        for _, p in ipairs(char:GetDescendants()) do
            if p:IsA("BasePart") then p.CanCollide = false end
        end

        local function getBehindCF()
            if not mChar or not mRoot or not mRoot.Parent then return nil end
            local behindPos = (mRoot.CFrame * stickOffset).Position
            local aimTarget = mRoot.Position + Vector3.new(0, 0.8, 0)
            return CFrame.new(behindPos, aimTarget), aimTarget
        end

        -- Immediately glue behind them on frame 0
        local initialCF, initialTarget = getBehindCF()
        if initialCF then
            myRoot.CFrame = initialCF
            myRoot.AssemblyLinearVelocity = Vector3.zero
            myRoot.AssemblyAngularVelocity = Vector3.zero
        end

        -- Continuously glue our character behind the Murderer on EVERY frame!
        -- If Murderer spins around or walks, our character rotates dynamically with them!
        local stickConn
        stickConn = RunService.Heartbeat:Connect(function()
            if tick() - startTime >= duration or not mRoot or not mRoot.Parent or not myRoot or not myRoot.Parent then
                if stickConn then
                    stickConn:Disconnect()
                    stickConn = nil
                end
                return
            end
            local curCF = getBehindCF()
            if curCF then
                myRoot.CFrame = curCF
                myRoot.AssemblyLinearVelocity = Vector3.zero
                myRoot.AssemblyAngularVelocity = Vector3.zero
            end
        end)

        -- Allow a tiny fraction of a second for server sync, then fire point-blank into their back
        task.wait(0.03)
        local currentBehindCF, currentAimTarget = getBehindCF()
        if currentBehindCF and currentAimTarget then
            local originPos = currentBehindCF.Position
            local originCF = CFrame.new(originPos, currentAimTarget)
            
            -- Dual format remote dispatch for 100% point-blank hit
            shootRemote:FireServer(originCF, CFrame.new(currentAimTarget))
            shootRemote:FireServer(originCF, currentAimTarget)
        end

        -- Wait out the rest of the duration while remaining glued behind them
        local remainingTime = duration - (tick() - startTime)
        if remainingTime > 0 then
            task.wait(remainingTime)
        end

        if stickConn then
            stickConn:Disconnect()
            stickConn = nil
        end

        -- Snap back to original location safely
        myRoot.CFrame = oldCF
        myRoot.AssemblyLinearVelocity = Vector3.zero
        myRoot.AssemblyAngularVelocity = Vector3.zero
        
        WindUI:Notify({ 
            Title = "Combat", 
            Content = (currentLang == "TH" and "วาร์ปยิงด้านหลังฆาตกรสำเร็จ!" or "TP Shot Behind Murderer!"), 
            Icon = "crosshair" 
        })
    else
        -- SUPER ACCURATE AIM LOCK
        local originPos = myRoot.Position + Vector3.new(0, 1.4, 0) -- Shoulder height clears floors/railings!
        local predictedPos, targetCF, isWallBlocked = getPredictedTarget(mChar, originPos)
        if not predictedPos then return end

        if isWallBlocked then
            WindUI:Notify({
                Title = "Aim Lock",
                Content = (currentLang == "TH" and "⚠️ ฆาตกรอยู่หลังกำแพง (อาจโดนสิ่งกีดขวาง)!" or "⚠️ Murderer is behind a wall!"),
                Icon = "alert-circle"
            })
        end

        -- Snap Character & Camera towards target for 100% directional alignment
        pcall(function()
            local lookAtXZ = Vector3.new(predictedPos.X, myRoot.Position.Y, predictedPos.Z)

local function SkidFling(targetPlr)
    local Character = LocalPlayer.Character
    local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
    local RootPart = Humanoid and Humanoid.RootPart
    
    local TCharacter = targetPlr and targetPlr.Character
    local THumanoid = TCharacter and TCharacter:FindFirstChildOfClass("Humanoid")
    local TRootPart = THumanoid and THumanoid.RootPart
    local THead = TCharacter and TCharacter:FindFirstChild("Head")
    local Accessory = TCharacter and TCharacter:FindFirstChildOfClass("Accessory")
    local Handle = Accessory and Accessory:FindFirstChild("Handle")

    if not Character or not Humanoid or not RootPart or not TCharacter or not TRootPart then 
        WindUI:Notify({ Title = "Fling", Content = (currentLang == "TH" and "ไม่พบเป้าหมาย หรือตัวละครไม่ถูกต้อง!" or "Target not found or invalid!"), Icon = "alert-circle" })
        return 
    end

    local oldPos = RootPart.CFrame
    local oldFPDH = workspace.FallenPartsDestroyHeight

    isFlinging = true
    for _, part in ipairs(Character:GetChildren()) do
        if part:IsA("BasePart") then part.CanCollide = true end
    end

    workspace.CurrentCamera.CameraSubject = THead or Handle or THumanoid
    workspace.FallenPartsDestroyHeight = 0/0 

    local FPos = function(BasePart, Pos, Ang)
        RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
        Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
        RootPart.Velocity = Vector3.new(9e7, 9e7 * 10, 9e7)
        RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
    end

    local SFBasePart = function(BasePart)
        local TimeToWait = 2
        local Time = tick()
        local Angle = 0
        repeat
            if RootPart and THumanoid and BasePart.Parent then
                if BasePart.Velocity.Magnitude < 50 then
                    Angle = Angle + 100
                    FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle),0 ,0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                    task.wait()
                else
                    FPos(BasePart, CFrame.new(0, 1.5, THumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, -THumanoid.WalkSpeed), CFrame.Angles(0, 0, 0))
                    task.wait()
                end
            end
        until Time + TimeToWait < tick()
    end

    local BV = Instance.new("BodyVelocity")
    BV.Parent = RootPart
    BV.Velocity = Vector3.new(0, 0, 0)
    BV.MaxForce = Vector3.new(9e9, 9e9, 9e9)

    Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)

    if TRootPart then SFBasePart(TRootPart)
    elseif THead then SFBasePart(THead)
    elseif Handle then SFBasePart(Handle) end

    BV:Destroy()
    Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
    workspace.CurrentCamera.CameraSubject = Humanoid

    local timeout = tick() + 2
    repeat
        RootPart.CFrame = oldPos * CFrame.new(0, 0.5, 0)
        Character:SetPrimaryPartCFrame(oldPos * CFrame.new(0, 0.5, 0))
        Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        for _, part in pairs(Character:GetChildren()) do
            if part:IsA("BasePart") then
                part.Velocity = Vector3.new()
                part.RotVelocity = Vector3.new()
            end
        end
        task.wait()
    until (RootPart.Position - oldPos.p).Magnitude < 25 or tick() > timeout

    workspace.FallenPartsDestroyHeight = oldFPDH
    isFlinging = false
    WindUI:Notify({ Title = "Fling", Content = (currentLang == "TH" and "ชนกระเด็นสำเร็จ!" or "Fling completed!"), Icon = "check" })
end

local function getMap()
    for _, obj in ipairs(workspace:GetChildren()) do

                end
            end
        end
    end
    return nil
end

-- SERVER HOP ENGINE (ACTIVE / POPULATED SERVERS)
--==================================================
local function serverHopClosest()
    WindUI:Notify({
        Title = "Server Hop",
        Content = (currentLang == "TH" and "กำลังค้นหาเซิร์ฟเวอร์ที่มีผู้เล่นปกติ..." or "Searching for populated active server..."),
        Icon = "refresh-cw"
    })

    task.spawn(function()
        local placeId = game.PlaceId
        local currentJobId = game.JobId
        local foundServer = nil

        local function fetchServers()
            -- Use sortOrder=Desc to get active, populated game rooms instead of ghost servers
            local url = "https://games.roblox.com/v1/games/" .. tostring(placeId) .. "/servers/Public?sortOrder=Desc&limit=100"
            local success, res = pcall(function()
                return game:HttpGetAsync(url)
            end)
            if not success or not res then
                success, res = pcall(function()
                    if request then
                        return request({Url = url, Method = "GET"}).Body
                    elseif syn and syn.request then
                        return syn.request({Url = url, Method = "GET"}).Body
                    elseif http and http.request then
                        return http.request({Url = url, Method = "GET"}).Body
                    end
                end)
            end
            if success and res then
                local s, decoded = pcall(function() return HttpService:JSONDecode(res) end)
                if s and decoded and decoded.data then
                    return decoded.data
                end
            end
            return nil
        end

        local servers = fetchServers()
        if servers and #servers > 0 then
            local goodServers = {}
            local fallbackServers = {}

            for _, server in ipairs(servers) do
                local playing = tonumber(server.playing) or 0
                local maxP = tonumber(server.maxPlayers) or 12
                if server.id ~= currentJobId and playing < maxP then
                    -- Normal active servers: between 5 and maxP-1 players
                    if playing >= 5 and playing <= (maxP - 1) then
                        table.insert(goodServers, server)
                    elseif playing > 1 and playing <= (maxP - 1) then
                        table.insert(fallbackServers, server)
                    end
                end
            end

            -- Pick randomly from good populated servers so we don't land in empty rooms
            if #goodServers > 0 then
                foundServer = goodServers[math.random(1, #goodServers)]
            elseif #fallbackServers > 0 then
                foundServer = fallbackServers[math.random(1, #fallbackServers)]
            end
        end

        if foundServer then
            local pCount = tostring(foundServer.playing or "?")
            local mCount = tostring(foundServer.maxPlayers or 12)
            WindUI:Notify({
                Title = "Server Hop",
                Content = (currentLang == "TH" and ("พบห้องปกติ (" .. pCount .. "/" .. mCount .. " คน) กำลังเชื่อมต่อ...") or ("Connecting to populated server (" .. pCount .. "/" .. mCount .. " players)...")),
                Icon = "check-circle"
            })
            task.wait(0.6)
            TeleportService:TeleportToPlaceInstance(placeId, foundServer.id, LocalPlayer)
        else
            WindUI:Notify({
                Title = "Server Hop",
                Content = (currentLang == "TH" and "กำลังย้ายเข้าเซิร์ฟเวอร์สาธารณะ..." or "Teleporting to public server..."),
                Icon = "refresh-cw"
            })
            task.wait(0.5)
            pcall(function()
                TeleportService:Teleport(placeId, LocalPlayer)
            end)
        end
    end)
end

--==================================================
-- TOGGLE EMOTE ENGINE (TRIPLE-STAGE + TOGGLE ON/OFF)
--==================================================
local currentEmoteTrack = nil
local currentEmoteId = nil

local function playEmote(id)
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then return end

    local animator = hum:FindFirstChildOfClass("Animator") or Instance.new("Animator", hum)
    local strId = tostring(id)

    if currentEmoteId == strId and currentEmoteTrack and currentEmoteTrack.IsPlaying then
        currentEmoteTrack:Stop()
        currentEmoteTrack = nil
        currentEmoteId = nil
        return
    end

    if currentEmoteTrack then
        currentEmoteTrack:Stop()
        currentEmoteTrack = nil
        currentEmoteId = nil
    end

    local numId = tonumber(id)
    local playedTrack = nil

    if not playedTrack then
        pcall(function()
            local objs = game:GetObjects("rbxassetid://" .. strId)
            if objs and #objs > 0 then
                for _, obj in ipairs(objs) do
                    local animObj = obj:IsA("Animation") and obj or obj:FindFirstChildWhichIsA("Animation", true)
                    if animObj then
                        playedTrack = animator:LoadAnimation(animObj)
                        break
                    end
                end
            end
        end)
    end

    if not playedTrack and numId then
        pcall(function()
            if hum.PlayEmoteAndGetAnimTrackById then
                playedTrack = hum:PlayEmoteAndGetAnimTrackById(numId)
            end
        end)
    end

    if not playedTrack then
        pcall(function()
            local anim = Instance.new("Animation")
            anim.AnimationId = "rbxassetid://" .. strId
            playedTrack = animator:LoadAnimation(anim)
        end)
    end

    if playedTrack then
        for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
            if track ~= playedTrack then
                track:Stop()
            end
        end
        playedTrack.Priority = Enum.AnimationPriority.Action4
        playedTrack:Play()
        currentEmoteTrack = playedTrack
        currentEmoteId = strId

local function resolveAnimationId(assetId)
    local rawId = "rbxassetid://" .. tostring(assetId)
    pcall(function()
        local objs = game:GetObjects("rbxassetid://" .. tostring(assetId))
        if objs and #objs > 0 then
            for _, obj in ipairs(objs) do
                if obj:IsA("Animation") and obj.AnimationId ~= "" then
                    rawId = obj.AnimationId
                    return
                end
                local found = obj:FindFirstChildWhichIsA("Animation", true)
                if found and found.AnimationId ~= "" then
                    rawId = found.AnimationId
                    return
                end
            end
        end
    end)
    return rawId
end

local function backupOriginalAnimations()
    local char = LocalPlayer.Character
    local animate = char and char:FindFirstChild("Animate")
    if not animate or originalAnimations then return end

    originalAnimations = {}
    for _, child in ipairs(animate:GetChildren()) do
        if child:IsA("StringValue") or child:IsA("Folder") then
            originalAnimations[child.Name] = {}
            for _, anim in ipairs(child:GetChildren()) do
                if anim:IsA("Animation") then
                    originalAnimations[child.Name][anim.Name] = anim.AnimationId
                end
            end
        end
    end
end

local function applyCustomAnimationPack(animMap)
    local char = LocalPlayer.Character
    local animate = char and char:FindFirstChild("Animate")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not animate or not hum then return end

    backupOriginalAnimations()

    for folderName, rawVal in pairs(animMap) do
        local folder = animate:FindFirstChild(folderName)
        if folder then
            local animId = (typeof(rawVal) == "string" and rawVal:find("rbxassetid://")) and rawVal or resolveAnimationId(rawVal)
            for _, anim in ipairs(folder:GetChildren()) do
                if anim:IsA("Animation") then
                    anim.AnimationId = animId
                end
            end
        end
    end

    local animator = hum:FindFirstChildOfClass("Animator")
    if animator then
        for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
            pcall(function() track:Stop(0.05) end)
        end
    end

    -- Reload Animate LocalScript so new pack plays instantly with no frozen T-pose!
    pcall(function()
        animate.Disabled = true
        task.wait(0.03)
        animate.Disabled = false
    end)
end

local function applyAuraAnimationPack()

        local animId = resolvedAuraPack[folderName]
        if not animId then
            animId = resolveAnimationId(assetId)
            resolvedAuraPack[folderName] = animId

    end
    applyCustomAnimationPack(resolvedAuraPack)
end

local function inspectAndResolveAsset(idStr)
    local cleanId = tostring(idStr):match("(%d+)")
    if not cleanId then return false, {}, nil end

    local isPack = false
    local packMap = {}
    local resolvedName = nil

    -- 1. Try Bundle API (Handles Roblox bundle URLs and Bundle IDs e.g. /bundles/83)
    pcall(function()
        local bUrl = "https://catalog.roblox.com/v1/bundles/" .. cleanId .. "/details"
        local res = nil
        if game.HttpGetAsync then
            res = game:HttpGetAsync(bUrl)
        elseif request then
            res = request({Url = bUrl, Method = "GET"}).Body
        elseif syn and syn.request then
            res = syn.request({Url = bUrl, Method = "GET"}).Body
        elseif http and http.request then
            res = http.request({Url = bUrl, Method = "GET"}).Body
        end

        if res then
            local decoded = HttpService:JSONDecode(res)
            if decoded and decoded.name then
                resolvedName = decoded.name
                local bundleType = decoded.bundleType or ""
                local items = decoded.items or {}
                local animTypes = {"idle", "walk", "run", "jump", "fall", "climb", "swim"}
                local matchCount = 0

                for _, itm in ipairs(items) do
                    local nLower = tostring(itm.name):lower()
                    for _, aType in ipairs(animTypes) do
                        if nLower:find(aType) then
                            packMap[aType] = tostring(itm.id)
                            matchCount = matchCount + 1
                            break
                        end
                    end
                end

                if matchCount >= 2 or bundleType:lower():find("animation") then
                    isPack = true
                end
            end
        end
    end)

    -- 2. Try AssetService:GetBundleDetailsAsync
    if not isPack then
        pcall(function()
            local AssetService = game:GetService("AssetService")
            local bDetails = AssetService:GetBundleDetailsAsync(tonumber(cleanId))
            if bDetails and bDetails.Name then
                resolvedName = bDetails.Name
                local animTypes = {"idle", "walk", "run", "jump", "fall", "climb", "swim"}
                local matchCount = 0
                for _, itm in ipairs(bDetails.Items or {}) do
                    local nLower = tostring(itm.Name):lower()
                    for _, aType in ipairs(animTypes) do
                        if nLower:find(aType) then
                            packMap[aType] = tostring(itm.Id)
                            matchCount = matchCount + 1
                            break
                        end
                    end
                end
                if matchCount >= 2 or tostring(bDetails.BundleType):lower():find("animation") then
                    isPack = true
                end
            end
        end)
    end

    -- 3. Try game:GetObjects (Handles Asset Packages with animation tracks)
    if not isPack then
        pcall(function()
            local objs = game:GetObjects("rbxassetid://" .. cleanId)
            if objs and #objs > 0 then
                local animTypes = {"idle", "walk", "run", "jump", "fall", "climb", "swim"}
                local foundCount = 0
                for _, obj in ipairs(objs) do
                    for _, aName in ipairs(animTypes) do
                        local found = obj:FindFirstChild(aName, true) or (obj.Name:lower():find(aName) and obj)
                        if found then
                            foundCount = foundCount + 1
                            local animObj = found:IsA("Animation") and found or found:FindFirstChildWhichIsA("Animation", true)
                            if animObj and animObj.AnimationId ~= "" then
                                packMap[aName] = animObj.AnimationId
                            end
                        end
                    end
                end
                if foundCount >= 2 then
                    isPack = true
                end
            end
        end)
    end

    return isPack, packMap, resolvedName
end

local function toggleCollapse()
    local mainFrame = Window.UIElements and Window.UIElements.Main
    if not mainFrame or isToggling then return end

    isCollapsed = not isCollapsed
    local sidebar = Window.UIElements.SideBarContainer
    local mainBar = Window.UIElements.MainBar
    local bg = mainFrame:FindFirstChild("Background")
    local innerMain = mainFrame:FindFirstChild("Main")

    if mainFrame.AnchorPoint.Y ~= 0 then
        local currentH = mainFrame.Size.Y.Offset > 0 and mainFrame.Size.Y.Offset or savedExpandedHeight
        mainFrame.Position = UDim2.new(
            mainFrame.Position.X.Scale,
            mainFrame.Position.X.Offset,
            mainFrame.Position.Y.Scale,
            mainFrame.Position.Y.Offset - (currentH * mainFrame.AnchorPoint.Y)
        )
        mainFrame.AnchorPoint = Vector2.new(0.5, 0)
    end

    local widthScale = mainFrame.Size.X.Scale
    local widthOffset = mainFrame.Size.X.Offset

    if isCollapsed then
        if mainFrame.Size.Y.Offset > TOPBAR_HEIGHT then
            savedExpandedHeight = mainFrame.Size.Y.Offset
        end

        mainFrame.ClipsDescendants = true
        if bg then bg.ClipsDescendants = true end
        if innerMain then innerMain.ClipsDescendants = true end

        if sidebar then sidebar.Visible = false end
        if mainBar then mainBar.Visible = false end
        mainFrame.Visible = true
        if innerMain then innerMain.Visible = true end

        local tween = TweenService:Create(mainFrame, TweenInfo.new(0.24, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
            Size = UDim2.new(widthScale, widthOffset, 0, TOPBAR_HEIGHT)
        })
        tween:Play()
        task.delay(0.25, function() isToggling = false end)
    else
        mainFrame.Visible = true
        if innerMain then innerMain.Visible = true end

        local targetHeight = savedExpandedHeight or defaultHeight

        if sidebar then sidebar.Visible = true end
        if mainBar then mainBar.Visible = true end

        if Window.TabModule and Window.TabModule.SelectedTab then

                Window.TabModule:SelectTab(Window.TabModule.SelectedTab)

            Size = UDim2.new(widthScale, widthOffset, 0, targetHeight)
        })

                mainFrame.ClipsDescendants = false
                if bg then bg.ClipsDescendants = false end
                if innerMain then innerMain.ClipsDescendants = false end

-- Smooth Draggable Container (Touch & Mouse Support)
QuickSkills.isDragging = false

local function onQuickDragBegan(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        QuickSkills.isDragging = true
        QuickSkills.dragStart = input.Position
        QuickSkills.startPos = QuickSkills.Container.Position
    end
end

QuickSkills.Container.InputBegan:Connect(onQuickDragBegan)
DragGrip.InputBegan:Connect(onQuickDragBegan)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        QuickSkills.isDragging = false
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if QuickSkills.isDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - QuickSkills.dragStart
        QuickSkills.Container.Position = UDim2.new(
            QuickSkills.startPos.X.Scale,
            QuickSkills.startPos.X.Offset + delta.X,
            QuickSkills.startPos.Y.Scale,
            QuickSkills.startPos.Y.Offset + delta.Y
        )
    end
end)

-- Micro-scale click animation
local function animateQuickButton(btn)
    if not btn then return end
    pcall(function()
        local orig = UDim2.fromOffset(54, 54)
        local pressed = UDim2.fromOffset(46, 46)
        TweenService:Create(btn, TweenInfo.new(0.08, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), { Size = pressed }):Play()
        task.delay(0.09, function()
            TweenService:Create(btn, TweenInfo.new(0.12, Enum.EasingStyle.Back, Enum.EasingDirection.Out), { Size = orig }):Play()
        end)
    end)
end

local function renderCustomItemButton(entry, index)
    local btn = TabPlayer:Button({
        Title = entry.Name .. (entry.IsPack and " 📦 [Pack]" or ""),
        Callback = function()
            if entry.IsPack and entry.PackMap then
                if activeCustomPackId == tostring(entry.Id) then
                    activeCustomPackId = nil
                    isAuraPackEnabled = false
                    restoreDefaultAnimationPack()
                    WindUI:Notify({ Title = "Animation Pack", Content = (currentLang == "TH" and "คืนค่าชุดท่าทางเริ่มต้นเรียบร้อย!" or "Restored Default Animations!"), Icon = "rotate-ccw" })
                else
                    activeCustomPackId = tostring(entry.Id)
                    isAuraPackEnabled = false
                    applyCustomAnimationPack(entry.PackMap)
                    WindUI:Notify({ Title = "Animation Pack", Content = (currentLang == "TH" and ("เปิดใช้งานชุดท่าทาง '" .. entry.Name .. "' สำเร็จ!") or ("Applied '" .. entry.Name .. "' Pack!")), Icon = "sparkles" })
                end
            else
                playEmote(entry.Id)
            end
        end,
    })
    customEmoteButtons[tostring(entry.Id)] = btn
    
    if entry.IsPack then
        setElementLayoutOrder(btn, 21 + (index or #customEmotes))
    else
        setElementLayoutOrder(btn, 40 + (index or #customEmotes))
    end
    return btn
end

local function reorderAllCustomItemButtons()
    local packCount = 0
    local emoteCount = 0
    if type(customEmotes) == "table" then
        for _, entry in ipairs(customEmotes) do
            local btn = customEmoteButtons[tostring(entry.Id)]
            if btn then
                if entry.IsPack then
                    packCount = packCount + 1
                    setElementLayoutOrder(btn, 21 + packCount)
                else
                    emoteCount = emoteCount + 1
                    setElementLayoutOrder(btn, 40 + emoteCount)
                end
            end
        end
    end
end

if type(customEmotes) == "table" then
    for idx, entry in ipairs(customEmotes) do
        renderCustomItemButton(entry, idx)
    end
end

local function getFormattedPlayerList()
    local formattedList = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            local role = getRole(p)
            local roleEmoji = "🛡️"
            if role == "Murderer" then roleEmoji = "🔪"
            elseif role == "Sheriff" then roleEmoji = "🔫"
            elseif role == "Hero" then roleEmoji = "⭐"
            elseif role == "Dead" then roleEmoji = "💀" end

            local displayTag = string.format("%s %s", roleEmoji, p.DisplayName)
            table.insert(formattedList, displayTag)
        end
    end
    return formattedList
end

local function parseTargetPlayerName(selectedStr)
    if not selectedStr then return nil end
    local clean = selectedStr:gsub("^%S+%s*", ""):match("^%s*(.-)%s*$")
    for _, p in ipairs(Players:GetPlayers()) do
        if p.DisplayName == clean or p.Name == clean or selectedStr:find(p.Name, 1, true) or selectedStr:find(p.DisplayName, 1, true) then
            return p.Name
        end
    end
    return clean
end

RunService.Stepped:Connect(function()
    if NoclipEnabled and not isFlinging and not isDropkicking and LocalPlayer.Character then
        for _, v in ipairs(LocalPlayer.Character:GetDescendants()) do
            if v:IsA("BasePart") and v.CanCollide then v.CanCollide = false end
        end
    end
end)

RunService.RenderStepped:Connect(function()
    if Destroyed then return end
    if FlyEnabled and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
        local hrp = LocalPlayer.Character.HumanoidRootPart
        local cam = workspace.CurrentCamera
        if not bg or not bv or bg.Parent ~= hrp or bv.Parent ~= hrp then startFly() end
        bg.cframe = cam.CFrame
        
        local PlayerModule = require(LocalPlayer:WaitForChild("PlayerScripts"):WaitForChild("PlayerModule"))
        local moveVector = PlayerModule:GetControls():GetMoveVector()
        
        local up = UserInputService:IsKeyDown(Enum.KeyCode.Space) and 1 or 0
        local down = UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) and -1 or 0
        local moveDir = Vector3.new(moveVector.X, up + down, moveVector.Z)
        
        if moveDir.Magnitude > 0 then bv.velocity = cam.CFrame:VectorToWorldSpace(moveDir) * flySpeed
        else bv.velocity = Vector3.zero end
    else
        if bg or bv then stopFly() end
    end
end)

task.spawn(function()
    while not Destroyed do
        task.wait(0.2)
        if ESPEnabled then refreshAllESP() end

        local hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid")
        if hum then
            if SpeedEnabled then
                hum.WalkSpeed = 25
                hum.JumpPower = 60
                hum.UseJumpPower = true
            else
                if hum.WalkSpeed > 20 then hum.WalkSpeed = 16 end
                if hum.JumpPower > 50 then hum.JumpPower = 50 end
            end
        end
    end
end)
