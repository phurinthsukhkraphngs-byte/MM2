local ESPFolder = Instance.new("Folder")

local function getRole(player)
    local role = "Innocent"
    if PlayerRoles[player.Name] then
        if PlayerRoles[player.Name].Dead then return "Dead" end
        role = PlayerRoles[player.Name].Role
    end

    local hasKnife, hasGun = false, false
    local function scanTools(parent)
        if not parent then return end
        for _, obj in ipairs(parent:GetChildren()) do
            if obj:IsA("Tool") then
                local n = obj.Name:lower()
                if n:find("knife") or n:find("stab") or obj:FindFirstChild("KnifeClient") then hasKnife = true
                elseif n:find("gun") or n:find("revolver") or n:find("pistol") then hasGun = true end
            end
        end
    end

    scanTools(player.Character)
    scanTools(player:FindFirstChildOfClass("Backpack"))

    if hasKnife then return "Murderer"
    elseif hasGun then return (role == "Sheriff" and "Sheriff" or "Hero") end
    return role
end

local function removeESP(player)
    local esp = ESPFolder:FindFirstChild(player.Name)
    if esp then esp:Destroy() end
end

local function updateESP(player)
    if Destroyed or player == LocalPlayer then return end
    local character = player.Character
    if not character or not character:FindFirstChild("Head") then removeESP(player) return end

    local role = getRole(player)
    if role == "Dead" then removeESP(player) return end 

    local color = COLORS[role] or Color3.new(1, 1, 1)
    local folder = ESPFolder:FindFirstChild(player.Name)

    if not folder then
        folder = Instance.new("Folder")
        folder.Name = player.Name
        folder.Parent = ESPFolder

        local highlight = Instance.new("Highlight")
        highlight.Name = "HL"
        highlight.Adornee = character
        highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        highlight.FillTransparency = 0.35
        highlight.OutlineTransparency = 0
        highlight.Parent = folder

        local billboard = Instance.new("BillboardGui")
        billboard.Name = "BB"
        billboard.Adornee = character.Head
        billboard.Size = UDim2.fromOffset(180, 40)
        billboard.StudsOffset = Vector3.new(0, 3, 0)
        billboard.AlwaysOnTop = true
        billboard.Parent = folder

        local text = Instance.new("TextLabel")
        text.Name = "TL"
        text.BackgroundTransparency = 1
        text.Size = UDim2.fromScale(1, 1)
        text.Font = Enum.Font.GothamBold
        text.TextSize = 14
        text.TextStrokeTransparency = 0
        text.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
        text.Parent = billboard
    end

    local hl = folder:FindFirstChild("HL")
    local bb = folder:FindFirstChild("BB")
    if hl and bb then
        hl.Adornee = character
        hl.FillColor = color
        hl.OutlineColor = color
        bb.Adornee = character:FindFirstChild("Head")

        if tl then
            tl.TextColor3 = color

                tl.Text = player.DisplayName .. " [" .. role .. "]"
            else
                tl.Text = player.DisplayName
            end

    end
end

local function getShootRemote()

local function getPredictedTarget(character, originPos)
    local root = character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Torso") or character:FindFirstChild("Head")
    if not root then return nil, nil end

    local hum = character:FindFirstChildOfClass("Humanoid")
    local vel = root.AssemblyLinearVelocity or root.Velocity or Vector3.zero

    if vel.Magnitude > 80 and hum then
        vel = hum.MoveDirection * hum.WalkSpeed
    end

    local ping = 0.05
    pcall(function()
        local pingStat = LocalPlayer:GetNetworkPing()
        if pingStat and pingStat > 0 then ping = pingStat end
    end)

    local dist = (root.Position - originPos).Magnitude
    local bulletSpeed = 450
    local timeToHit = (dist / bulletSpeed) + ping

    local predictedPos = root.Position + (vel * timeToHit)
    predictedPos = Vector3.new(predictedPos.X, root.Position.Y + 0.5, predictedPos.Z)

    local targetCF = CFrame.new(predictedPos, predictedPos + (vel.Magnitude > 0 and vel or root.CFrame.LookVector))
    return predictedPos, targetCF
end

    local murderer = findRole("Murderer")
    if not murderer or not murderer.Character then
        WindUI:Notify({ Title = "Combat", Content = "No Murderer found alive!", Icon = "alert-triangle" })
        return
    end
    
    local mChar = murderer.Character
    local mRoot = mChar:FindFirstChild("HumanoidRootPart") or mChar:FindFirstChild("Torso") or mChar:FindFirstChild("Head")
    if not mRoot then return end

    local char = LocalPlayer.Character
    local myRoot = char and char:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end

    local gun = char:FindFirstChild("Gun") or LocalPlayer.Backpack:FindFirstChild("Gun")
    if not gun then
        WindUI:Notify({ Title = "Combat", Content = "Equip Gun first!", Icon = "alert-circle" })
        return
    end

    local hum = char:FindFirstChildOfClass("Humanoid")
    if gun.Parent == LocalPlayer.Backpack and hum then
        hum:EquipTool(gun)
        local t = tick() + 0.5
        repeat task.wait() until char:FindFirstChild("Gun") or tick() > t
    end

    local shootRemote = getShootRemote()
    if not shootRemote then return end

    local oldCF = myRoot.CFrame

    if teleport then
        local tpPos = mRoot.CFrame * CFrame.new(0, 1, 2)
        myRoot.CFrame = tpPos
        myRoot.AssemblyLinearVelocity = Vector3.zero
        task.wait(0.04)

        local freshTargetPos = mRoot.Position + Vector3.new(0, 0.5, 0)
        local originCF = CFrame.new(myRoot.Position, freshTargetPos)
        shootRemote:FireServer(originCF, CFrame.new(freshTargetPos))

        task.wait(0.12)
        myRoot.CFrame = oldCF
        myRoot.AssemblyLinearVelocity = Vector3.zero
        WindUI:Notify({ Title = "Combat", Content = "TP Shot Murderer!", Icon = "crosshair" })
    else
        local predictedPos, targetCF = getPredictedTarget(mChar, myRoot.Position)
        if not predictedPos then return end

        local originCF = CFrame.new(myRoot.Position, predictedPos)
        shootRemote:FireServer(originCF, targetCF)
        WindUI:Notify({ Title = "Combat", Content = "Aim Locked Murderer!", Icon = "crosshair" })
    end
end

local function executeKnifeThrowNearest()
    local char = LocalPlayer.Character
    local myRoot = char and char:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end

    local target = nil
    local shortestDist = math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("Head") then
            if getRole(p) ~= "Dead" then
                local dist = (p.Character.Head.Position - myRoot.Position).Magnitude
                if dist < shortestDist then
                    shortestDist = dist
                    target = p
                end
            end
        end
    end

    if not target then 
        WindUI:Notify({ Title = "Combat", Content = "No alive targets nearby!", Icon = "alert-circle" })
        return 
    end

    local knife = char:FindFirstChild("Knife") or LocalPlayer.Backpack:FindFirstChild("Knife")
    if not knife then
        WindUI:Notify({ Title = "Combat", Content = "Equip Knife first!", Icon = "alert-circle" })
        return 
    end

    local hum = char:FindFirstChildOfClass("Humanoid")
    if knife.Parent == LocalPlayer.Backpack and hum then
        hum:EquipTool(knife) task.wait(0.1)
    end

    local events = knife:FindFirstChild("Events")
    local throwRemote = events and events:FindFirstChild("KnifeThrown")
    if not throwRemote then return end

    local predictedPos, targetCF = getPredictedTarget(target.Character, myRoot.Position)
    if not predictedPos then return end

    local originCF = CFrame.new(myRoot.Position, predictedPos)
    throwRemote:FireServer(originCF, targetCF)
    WindUI:Notify({ Title = "Combat", Content = "Knife thrown at " .. target.DisplayName, Icon = "zap" })
end

local function SkidFling(targetPlr)
    local Character = LocalPlayer.Character
    local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
    local RootPart = Humanoid and Humanoid.RootPart
    
    local TCharacter = targetPlr and targetPlr.Character
    local THumanoid = TCharacter and TCharacter:FindFirstChildOfClass("Humanoid")
    local TRootPart = THumanoid and THumanoid.RootPart
    local THead = TCharacter and TCharacter:FindFirstChild("Head")
    local Accessory = TCharacter and TCharacter:FindFirstChildOfClass("Accessory")
    local Handle = Accessory and Accessory:FindFirstChild("Handle")

    if not Character or not Humanoid or not RootPart or not TCharacter or not TRootPart then 
        WindUI:Notify({ Title = "Fling", Content = "Target not found or invalid!", Icon = "alert-circle" })
        return 
    end

    local oldPos = RootPart.CFrame
    local oldFPDH = workspace.FallenPartsDestroyHeight

    isFlinging = true
    for _, part in ipairs(Character:GetChildren()) do
        if part:IsA("BasePart") then part.CanCollide = true end
    end

    workspace.CurrentCamera.CameraSubject = THead or Handle or THumanoid
    workspace.FallenPartsDestroyHeight = 0/0 

    local FPos = function(BasePart, Pos, Ang)
        RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
        Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
        RootPart.Velocity = Vector3.new(9e7, 9e7 * 10, 9e7)
        RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
    end

    local SFBasePart = function(BasePart)
        local TimeToWait = 2
        local Time = tick()
        local Angle = 0
        repeat
            if RootPart and THumanoid and BasePart.Parent then
                if BasePart.Velocity.Magnitude < 50 then
                    Angle = Angle + 100
                    FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle),0 ,0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                    task.wait()
                else
                    FPos(BasePart, CFrame.new(0, 1.5, THumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, -THumanoid.WalkSpeed), CFrame.Angles(0, 0, 0))
                    task.wait()
                end
            end
        until Time + TimeToWait < tick()
    end

    local BV = Instance.new("BodyVelocity")
    BV.Parent = RootPart
    BV.Velocity = Vector3.new(0, 0, 0)
    BV.MaxForce = Vector3.new(9e9, 9e9, 9e9)

    Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)

    if TRootPart then SFBasePart(TRootPart)
    elseif THead then SFBasePart(THead)
    elseif Handle then SFBasePart(Handle) end

    BV:Destroy()
    Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
    workspace.CurrentCamera.CameraSubject = Humanoid

    local timeout = tick() + 2
    repeat
        RootPart.CFrame = oldPos * CFrame.new(0, 0.5, 0)
        Character:SetPrimaryPartCFrame(oldPos * CFrame.new(0, 0.5, 0))
        Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        for _, part in pairs(Character:GetChildren()) do
            if part:IsA("BasePart") then
                part.Velocity = Vector3.new()
                part.RotVelocity = Vector3.new()
            end
        end
        task.wait()
    until (RootPart.Position - oldPos.p).Magnitude < 25 or tick() > timeout

    workspace.FallenPartsDestroyHeight = oldFPDH
    isFlinging = false
    WindUI:Notify({ Title = "Fling", Content = "Fling completed!", Icon = "check" })
end

local DropkickAnim = Instance.new("Animation")
DropkickAnim.AnimationId = "rbxassetid://133566007754001"

local function executeDropkickFling()
    if isDropkicking then return end
    local char = LocalPlayer.Character
    if not char then return end
    
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    if hum.RigType ~= Enum.HumanoidRigType.R15 then 
        WindUI:Notify({ Title = "Dropkick", Content = "Character must be R15 Rig!", Icon = "alert-triangle" })
        return 
    end

    local leg = char:FindFirstChild("LeftLowerLeg")
    if not leg then return end

    local animator = hum:FindFirstChildOfClass("Animator") or Instance.new("Animator", hum)
    local s, track = pcall(function() return animator:LoadAnimation(DropkickAnim) end)
    if not s or not track then return end

    isDropkicking = true
    WindUI:Notify({ Title = "Dropkick", Content = "Dropkick Fling triggered!", Icon = "zap" })

    task.spawn(function()
        local noclipConn = RunService.Stepped:Connect(function()
            if char and char.Parent then
                for _, child in ipairs(char:GetDescendants()) do
                    if child:IsA("BasePart") and child.CanCollide then child.CanCollide = false end
                end
            end
        end)

        pcall(function()
            track:Play()

            local oldvelocity = leg.Velocity
            local originalHipHeight = hum.HipHeight

            repeat
                RunService.Heartbeat:Wait()
                hum.HipHeight = 2.5008
                local velocity, movel = leg.Velocity, 0.1
                leg.Velocity = velocity * 1000000 + Vector3.new(0, 1000000, 0)

                RunService.RenderStepped:Wait()
                leg.Velocity = velocity

                RunService.Stepped:Wait()
                leg.Velocity = velocity + Vector3.new(0, movel, 0)
            until track.IsPlaying == false or not char.Parent

            hum.HipHeight = (originalHipHeight > 0) and originalHipHeight or 2.0008
            leg.Velocity = oldvelocity
        end)

        if noclipConn then noclipConn:Disconnect() end
        isDropkicking = false
    end)
end

local AntiFlingConn = nil
local function toggleAntiFling(val)
    AntiFlingEnabled = val
    if AntiFlingConn then AntiFlingConn:Disconnect() AntiFlingConn = nil end

    if AntiFlingEnabled then
        AntiFlingConn = RunService.Stepped:Connect(function()
            if not AntiFlingEnabled or isFlinging or isDropkicking then return end
            local myChar = LocalPlayer.Character
            local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
            local myHum = myChar and myChar:FindFirstChildOfClass("Humanoid")

            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character then
                    for _, part in ipairs(p.Character:GetChildren()) do
                        if part:IsA("BasePart") and part.CanCollide then
                            part.CanCollide = false
                        end
                    end
                end
            end

            if myHrp and not FlyEnabled then
                local linVel = myHrp.AssemblyLinearVelocity or myHrp.Velocity
                local rotVel = myHrp.AssemblyAngularVelocity or myHrp.RotVelocity

                if linVel.Magnitude > 85 then
                    myHrp.AssemblyLinearVelocity = Vector3.zero
                    myHrp.Velocity = Vector3.zero
                end

                if rotVel.Magnitude > 85 then
                    myHrp.AssemblyAngularVelocity = Vector3.zero
                    myHrp.RotVelocity = Vector3.zero
                end
            end

            if myHum then
                myHum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
                myHum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
            end
        end)
    end
end

local function getMap()
    for _, obj in ipairs(workspace:GetChildren()) do

                if obj:FindFirstChild("CoinContainer") or obj:FindFirstChild("CoinServer") or obj:FindFirstChild("Spawns") then
                    return obj
                end
            end
        end
    end
    return nil
end

local function tpToLobby()
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local lobby = getLobby()
    if lobby then
        local spawns = lobby:FindFirstChild("Spawns", true) or lobby:FindFirstChild("SpawnPoints", true)
        if spawns then
            local validParts = {}
            for _, p in ipairs(spawns:GetChildren()) do
                if p:IsA("BasePart") then table.insert(validParts, p) end
            end
            if #validParts > 0 then
                root.CFrame = validParts[math.random(1, #validParts)].CFrame * CFrame.new(0, 3, 0)
                WindUI:Notify({ Title = "Teleport", Content = "Teleported to Lobby!", Icon = "map-pin" })
                return
            end
        end

        local spawnLoc = lobby:FindFirstChildWhichIsA("SpawnLocation", true) or lobby:FindFirstChildWhichIsA("BasePart", true)
        if spawnLoc then
            root.CFrame = spawnLoc.CFrame * CFrame.new(0, 3, 0)
            WindUI:Notify({ Title = "Teleport", Content = "Teleported to Lobby!", Icon = "map-pin" })
            return
        end
    end
    WindUI:Notify({ Title = "Teleport", Content = "Lobby not found!", Icon = "alert-circle" })
end

local function tpToMap()
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local map = getMap()
    if not map then 
        WindUI:Notify({ Title = "Teleport", Content = "No active map in game round!", Icon = "alert-triangle" })
        return
    end

    local spawns = map:FindFirstChild("Spawns", true) or map:FindFirstChild("SpawnPoints", true)
    if spawns then
        local validParts = {}
        for _, p in ipairs(spawns:GetChildren()) do
            if p:IsA("BasePart") then table.insert(validParts, p) end
        end
        if #validParts > 0 then
            root.CFrame = validParts[math.random(1, #validParts)].CFrame * CFrame.new(0, 3, 0)
            WindUI:Notify({ Title = "Teleport", Content = "Teleported to Map Spawn!", Icon = "map-pin" })
            return
        end
    end

    local coins = map:FindFirstChild("CoinContainer", true)
    if coins then
        local coinPart = coins:FindFirstChildWhichIsA("BasePart", true)
        if coinPart then
            root.CFrame = coinPart.CFrame * CFrame.new(0, 3, 0)
            WindUI:Notify({ Title = "Teleport", Content = "Teleported to Map Coins!", Icon = "map-pin" })
            return
        end
    end

    for _, part in ipairs(map:GetDescendants()) do
        if part:IsA("BasePart") and part.CanCollide and not part.Name:lower():find("kill") and not part.Name:lower():find("fall") then
            root.CFrame = part.CFrame * CFrame.new(0, 3, 0)
            WindUI:Notify({ Title = "Teleport", Content = "Teleported to Map!", Icon = "map-pin" })
            return
        end
    end
end

local function playEmote(id)
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then return end

    local animator = hum:FindFirstChildOfClass("Animator") or Instance.new("Animator", hum)
    for _, track in ipairs(animator:GetPlayingAnimationTracks()) do track:Stop() end

    local anim = Instance.new("Animation")
    anim.AnimationId = "rbxassetid://" .. tostring(id)

    local success, track = pcall(function() return animator:LoadAnimation(anim) end)
    if success and track then
        track.Priority = Enum.AnimationPriority.Action
        track:Play()
    end
end

local function changeSkin(toolName, meshId, texId)
    local tool = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild(toolName)
    if not tool then tool = LocalPlayer.Backpack:FindFirstChild(toolName) end
    if tool then
        local handle = tool:FindFirstChild("Handle")

            local mesh = handle:FindFirstChildOfClass("SpecialMesh") or Instance.new("SpecialMesh", handle)
            mesh.MeshType = Enum.MeshType.FileMesh

            mesh.TextureId = texId and ("rbxassetid://" .. texId) or ""
            WindUI:Notify({ Title = "Skins", Content = "Applied " .. toolName .. " Skin!", Icon = "sparkles" })
        end
    end
end

-- ZERO-MOTION TOPBAR COLLAPSE ENGINE (TOP-ANCHORED)
--==================================================
local isCollapsed = false
local isToggling = false
local savedExpandedHeight = 440

-- ตั้งค่า AnchorPoint.Y ให้อยู่ที่ขอบบน (0) ตั้งแต่เริ่ม เพื่อให้บาร์บนล็อคอยู่กับที่ 100%
task.spawn(function()
    task.wait(0.1)
    local mainFrame = Window.UIElements and Window.UIElements.Main
    if mainFrame and mainFrame.AnchorPoint.Y ~= 0 then
        local currentH = mainFrame.Size.Y.Offset > 0 and mainFrame.Size.Y.Offset or 440
        mainFrame.Position = UDim2.new(
            mainFrame.Position.X.Scale,
            mainFrame.Position.X.Offset,
            mainFrame.Position.Y.Scale,
            mainFrame.Position.Y.Offset - (currentH * mainFrame.AnchorPoint.Y)
        )
        mainFrame.AnchorPoint = Vector2.new(0.5, 0)
    end
end)

local function toggleCollapse()
    local mainFrame = Window.UIElements and Window.UIElements.Main
    if not mainFrame or isToggling then return end
    isToggling = true

    isCollapsed = not isCollapsed
    local sidebar = Window.UIElements.SideBarContainer
    local mainBar = Window.UIElements.MainBar
    local bg = mainFrame:FindFirstChild("Background")
    local innerMain = mainFrame:FindFirstChild("Main")

    mainFrame.ClipsDescendants = true
    if bg then bg.ClipsDescendants = true end
    if innerMain then innerMain.ClipsDescendants = true end

    -- มั่นใจว่า AnchorPoint.Y เป็น 0 เสมอ
    if mainFrame.AnchorPoint.Y ~= 0 then
        local currentH = mainFrame.Size.Y.Offset > 0 and mainFrame.Size.Y.Offset or 440
        mainFrame.Position = UDim2.new(
            mainFrame.Position.X.Scale,
            mainFrame.Position.X.Offset,
            mainFrame.Position.Y.Scale,
            mainFrame.Position.Y.Offset - (currentH * mainFrame.AnchorPoint.Y)
        )
        mainFrame.AnchorPoint = Vector2.new(0.5, 0)
    end

    local widthScale = mainFrame.Size.X.Scale
    local widthOffset = mainFrame.Size.X.Offset

    if isCollapsed then
        if mainFrame.Size.Y.Offset > TOPBAR_HEIGHT then
            savedExpandedHeight = mainFrame.Size.Y.Offset
        end

        if sidebar then sidebar.Visible = false end
        if mainBar then mainBar.Visible = false end
        mainFrame.Visible = true
        if innerMain then innerMain.Visible = true end

        -- ปรับเฉพาะความสูง (Size) โดยไม่แตะต้อง Position เลยแม้แต่น้อย บาร์บนจึงอยู่นิ่งสนิท 100%
        local tween = TweenService:Create(mainFrame, TweenInfo.new(0.24, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
            Size = UDim2.new(widthScale, widthOffset, 0, TOPBAR_HEIGHT)
        })
        tween:Play()
        task.delay(0.25, function() isToggling = false end)
    else
        mainFrame.Visible = true
        if innerMain then innerMain.Visible = true end

        local targetHeight = savedExpandedHeight or 440

        -- ปรับเฉพาะความสูง (Size) คลี่ลงด้านล่าง โดยบาร์บนอยู่ที่เดิม 100% ไม่ขยับ
        local tween = TweenService:Create(mainFrame, TweenInfo.new(0.24, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
            Size = UDim2.new(widthScale, widthOffset, 0, targetHeight)
        })
        tween:Play()

        task.delay(0.16, function()
            if not isCollapsed then
                if sidebar then sidebar.Visible = true end
                if mainBar then mainBar.Visible = true end
            end
        end)
        task.delay(0.25, function() isToggling = false end)
    end
end

-- FULL UNLOAD FUNCTION (ทำลาย UI ทั้งหมดอย่างหมดจด 100%)
--==================================================
local function fullUnloadScript()
    Destroyed = true
    if ESPFolder then pcall(function() ESPFolder:Destroy() end) end
    if RestoreGui then pcall(function() RestoreGui:Destroy() end) end
    if AntiFlingConn then pcall(function() AntiFlingConn:Disconnect() end) end
    stopFly()

    -- ทำลาย ScreenGui ทั้งหมดของ WindUI
    pcall(function() if WindUI and WindUI.ScreenGui then WindUI.ScreenGui:Destroy() end end)
    pcall(function() if WindUI and WindUI.NotificationGui then WindUI.NotificationGui:Destroy() end end)
    pcall(function() if WindUI and WindUI.DropdownGui then WindUI.DropdownGui:Destroy() end end)
    pcall(function() if WindUI and WindUI.TooltipGui then WindUI.TooltipGui:Destroy() end end)

    -- กวาดล้าง UI ที่หลงเหลือทั้งหมด
    pcall(function()
        for _, obj in ipairs(game.CoreGui:GetChildren()) do
            if obj.Name == "R1N_MM2_ESP" or obj.Name == "R1N_RESTORE_BAR" or obj.Name:find("WindUI") then
                obj:Destroy()
            end
        end
    end)
end

-- Live Role Badge Updater Loop (Neon Vibrant Colors)
task.spawn(function()
    local lastRole = nil
    while not Destroyed do
        task.wait(0.4)
        local myRole = getRole(LocalPlayer)
        if myRole ~= lastRole then
            lastRole = myRole
            local tagColor = COLORS[myRole] or Color3.fromRGB(150, 150, 165)
            local tagIcon = "solar:user-bold"

            if myRole == "Murderer" then
                tagIcon = "solar:danger-bold"
            elseif myRole == "Sheriff" then
                tagIcon = "solar:shield-check-bold"
            elseif myRole == "Hero" then
                tagIcon = "solar:star-bold"
            elseif myRole == "Dead" then
                tagIcon = "solar:skull-bold"
            end

            pcall(function()
                RoleTag:SetTitle(myRole)
                RoleTag:SetColor(tagColor)
                RoleTag:SetIcon(tagIcon)
            end)
        end
    end
end)

task.spawn(function()
    local hue = 0
    while not Destroyed do
        task.wait(0.1)
        if ChromaEnabled then
            hue = hue + 0.05
            if hue > 1 then hue = 0 end
            local c = Color3.fromHSV(hue, 1, 1)
            local char = LocalPlayer.Character
            if char then
                for _, w in ipairs({"Knife", "Gun"}) do
                    local t = char:FindFirstChild(w)
                    if t and t:FindFirstChild("Handle") then
                        t.Handle.Material = Enum.Material.Neon
                        t.Handle.Color = c
                        local m = t.Handle:FindFirstChildOfClass("SpecialMesh")
                        if m then m.VertexColor = Vector3.new(c.R, c.G, c.B) end
                    end
                end
            end
        end
    end
end)

RunService.Stepped:Connect(function()
    if NoclipEnabled and not isFlinging and not isDropkicking and LocalPlayer.Character then
        for _, v in ipairs(LocalPlayer.Character:GetDescendants()) do
            if v:IsA("BasePart") and v.CanCollide then v.CanCollide = false end
        end
    end
end)

RunService.RenderStepped:Connect(function()
    if Destroyed then return end
    if FlyEnabled and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
        local hrp = LocalPlayer.Character.HumanoidRootPart
        local cam = workspace.CurrentCamera
        if not bg or not bv or bg.Parent ~= hrp or bv.Parent ~= hrp then startFly() end
        bg.cframe = cam.CFrame
        
        local PlayerModule = require(LocalPlayer:WaitForChild("PlayerScripts"):WaitForChild("PlayerModule"))
        local moveVector = PlayerModule:GetControls():GetMoveVector()
        
        local up = UserInputService:IsKeyDown(Enum.KeyCode.Space) and 1 or 0
        local down = UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) and -1 or 0
        local moveDir = Vector3.new(moveVector.X, up + down, moveVector.Z)
        
        if moveDir.Magnitude > 0 then bv.velocity = cam.CFrame:VectorToWorldSpace(moveDir) * flySpeed
        else bv.velocity = Vector3.zero end
    else
        if bg or bv then stopFly() end
    end
end)

task.spawn(function()
    while not Destroyed do
        task.wait(0.2)
        if ESPEnabled then refreshAllESP() end
        if AutoGetGunEnabled then pcall(function() tryGetGun(true) end) end

        local hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid")
        if hum then
            if SpeedEnabled then
                hum.WalkSpeed = 25
                hum.JumpPower = 60
                hum.UseJumpPower = true
            else
                if hum.WalkSpeed > 20 then hum.WalkSpeed = 16 end
                if hum.JumpPower > 50 then hum.JumpPower = 50 end
            end
        end
    end
end)
